function catOriginal(la,cz){
 let s=(la+' '+cz).toLowerCase();
 if(/fractura|zlomen|vulnus|rána|contusio|zhmožd|trauma|úraz|luxatio|distorsio/.test(s))return 'Úrazy';
 if(/status post|in statu|post operationem|propter|cum dyspnoe|cum dislocatione|post partum/.test(s))return 'Celá spojení a stavy';
 if(/gravid|uter|partus|porod|ovari|vagin|mamm|těhot/.test(s))return 'Porodnictví a gynekologie';
 if(/ectomia|stomia|operatio|resectio|chirurg|amputatio|resuscitatio/.test(s))return 'Operace a léčba';
 if(/arter|vena|venos|aorta|cor\b|cordis|srdc|tepna|žíla|chlopeň|atrium|ventricul.*cord/.test(s))return 'Srdce a cévy';
 if(/nerv|n\.\s|cerebr|encephal|medulla|moz|mícha|plexus|mening|arachnoid|neur|gyrus/.test(s))return 'Nervová soustava';
 if(/muscul|m\.\s|mm\.\s|sval|flexor|extensor|abductor|adductor|levator|depressor|pronat|supinat/.test(s))return 'Svaly a pohyby';
 if(/ligament|articul|kloub|vaz\b|tendo|šlacha/.test(s))return 'Klouby a vazy';
 if(/\bos\b|ossa|scapul|tibia|fibula|ulna|radius|humer|vertebr|kost|sternum|cranium|clavicul|costa|femur|patella/.test(s))return 'Kosti a kostra';
 if(/bronch|pulmon|trache|laryng|plíc|průduš/.test(s))return 'Dýchací soustava';
 if(/oesoph|gaster|ventricul.*žal|intestin|colon|rectum|hepar|pancre|žalud|střev|játr/.test(s))return 'Trávicí soustava';
 if(/itis\b|osis\b|oma\b|cancer|carcin|morbus|insufficien|paresis|plegia|nemoc|zánět|nádor|rakovina/.test(s))return 'Nemoci a diagnózy';
 if(/dolor|algia|bolest|phobia|krvácení|haemorrhag|epistax|spavost|insomn|vomit|zvracen|dušnost|dyspnoe/.test(s))return 'Příznaky';

}

function combine(){
 const m=new Map();
 for(const x of originals){
   if(PURE_GRAMMAR.test(x.la+' '+x.cz))continue;
   m.set(key(x),x);
 }
 for(const x of EXTRA){
   const y={...x,la:norm(x.la),cz:norm(x.cz),sources:Array.isArray(x.sources)?x.sources:['presentation']};
   const k=key(y);
   if(m.has(k)){
     const old=m.get(k);
     old.sources=[...new Set([...(old.sources||[]),...y.sources])];
     if(y.sources.includes('test'))old.category=y.category;
   }else m.set(k,y);
 }
 cards=[...m.values()].map((x,i)=>({...x,id:key(x)+'#'+i}));
 const cats=[...new Set(cards.map(x=>x.category))].sort((a,b)=>a.localeCompare(b,'cs'));
 if(settings.cats){

   selectedCats=new Set(valid.length?valid:cats);

}

async function load(){
 // Extras work immediately even if the original source cannot load.
 cards=EXTRA.map((x,i)=>({...x,id:key(x)+'#e'+i}));
 combine();
 try{
  const r=await fetch(BASE_URL,{cache:'no-store'});
  if(!r.ok)throw Error('HTTP '+r.status);
  const h=await r.text();
  const m=h.match(/const\s+WORDS\s*=\s*`([\s\S]*?)`/);
  if(!m)throw Error('nenalezena databáze WORDS');
  const rows=m[1].split(/\r?\n/).map(s=>s.trim()).filter(s=>s&&!s.startsWith('#'));
  originals=rows.map(line=>{
    const a=line.split(';').map(x=>x.trim());
    let la=a[0]||'',cz=a[1]||'';
    la=CORRECTIONS[la.toLowerCase()]||la;
    return {la,cz,category:catOriginal(la,cz),sources:['original']};
  }).filter(x=>x.la&&x.cz);
  $('load').className='notice ok';
  $('load').textContent='✓ Původní databáze načtena: '+originals.length+' položek. Navíc je v této stránce zabudováno '+EXTRA.length+' položek z prezentací/testů před sloučením duplicit.';
  combine();
 }catch(e){
  $('load').className='notice warn';
  $('load').textContent='⚠ Původních 805 se nepodařilo načíst ('+e.message+'). Prezentace a testy ale fungují: '+EXTRA.length+' zabudovaných položek. Obnov stránku při připojení k internetu.';
 }
}
load();
